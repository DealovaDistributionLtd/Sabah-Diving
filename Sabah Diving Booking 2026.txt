    <script>
        const CLIENT_ID = '346757037192-9vshmkrobprrcp64hn6c8diqlnuhhhhe.apps.googleusercontent.com';
        const CALENDAR_ID = 'bd0883e3e1e8410e121cd052d9e09c5ed0d1eee32ef9c71ebb6c8cafc354d221@group.calendar.google.com';
        const SCOPES = 'https://www.googleapis.com/auth/calendar.events';

        let tokenClient;
        let gapiInited = false;
        let gisInited = false;
        let isAuthorized = false;

        let bookings = JSON.parse(localStorage.getItem('sabah_dive_bookings')) || [];

        window.onload = function() {
            renderTable();
            gapi.load('client', initializeGapiClient);
            gisLoaded();
        };

        async function initializeGapiClient() {
            await gapi.client.init({
                discoveryDocs: ['https://www.googleapis.com/discovery/v1/apis/calendar/v3/rest'],
            });
            gapiInited = true;
        }

        function gisLoaded() {
            tokenClient = google.accounts.oauth2.initTokenClient({
                client_id: CLIENT_ID,
                scope: SCOPES,
                callback: (resp) => {
                    if (resp.error !== undefined) {
                        alert("Authentication failed.");
                        return;
                    }
                    isAuthorized = true;
                    const authBtn = document.getElementById('authButton');
                    authBtn.innerText = "✅ API Connected";
                    authBtn.classList.replace('bg-blue-600', 'bg-emerald-600');
                },
            });
            gisInited = true;
        }

        function handleAuthClick() {
            if (!gisInited || !gapiInited) {
                alert("Google API is still initializing. Please wait a second and try again.");
                return;
            }
            tokenClient.requestAccessToken({prompt: 'consent'});
        }

        async function addBooking(event) {
            event.preventDefault();
            
            const newBooking = {
                id: Date.now(),
                name: document.getElementById('guestName').value,
                passport: document.getElementById('passport').value,
                date: document.getElementById('activityDate').value,
                location: document.getElementById('location').value,
                package: document.getElementById('packageType').value,
                price: document.getElementById('totalPrice').value,
                status: 'Saved Locally'
            };

            if (isAuthorized) {
                const success = await pushEventToGoogleCalendar(newBooking);
                if (success) {
                    newBooking.status = '✅ Synced Live';
                }
            } else {
                alert("Booking saved locally! Click 'Connect Google Calendar' at the top to enable instant auto-syncing.");
            }

            bookings.push(newBooking);
            localStorage.setItem('sabah_dive_bookings', JSON.stringify(bookings));
            renderTable();
            document.getElementById('bookingForm').reset();
        }

        async function pushEventToGoogleCalendar(booking) {
            const event = {
                'summary': `Dive: ${booking.name} (${booking.package})`,
                'description': `Passport: ${booking.passport}\nLocation: ${booking.location}\nTotal Price: RM${booking.price}`,
                'start': { 'date': booking.date },
                'end': { 'date': booking.date }
            };

            try {
                const request = gapi.client.calendar.events.insert({
                    'calendarId': CALENDAR_ID,
                    'resource': event,
                });
                await request;
                return true;
            } catch (err) {
                console.error("API Push Error:", err);
                alert("⚠️ Saved locally, but calendar push failed. Please reconnect your Google Calendar account.");
                return false;
            }
        }

        function deleteBooking(id) {
            bookings = bookings.filter(b => b.id !== id);
            localStorage.setItem('sabah_dive_bookings', JSON.stringify(bookings));
            renderTable();
        }

        function renderTable() {
            const tbody = document.getElementById('bookingTableBody');
            tbody.innerHTML = '';
            if (bookings.length === 0) {
                tbody.innerHTML = `<tr><td colspan="7" class="py-4 text-center text-slate-500">No bookings recorded yet.</td></tr>`;
                return;
            }
            bookings.forEach(b => {
                tbody.innerHTML += `
                    <tr class="hover:bg-slate-700/30 transition">
                        <td class="py-3 px-2 font-medium text-slate-200">${b.name}</td>
                        <td class="py-3 px-2 text-slate-400">${b.passport}</td>
                        <td class="py-3 px-2 text-cyan-400">${b.date}</td>
                        <td class="py-3 px-2 text-slate-300">${b.location}</td>
                        <td class="py-3 px-2 text-slate-300">${b.package}</td>
                        <td class="py-3 px-2 text-xs font-semibold text-emerald-400">${b.status}</td>
                        <td class="py-3 px-2">
                            <button onclick="deleteBooking(${b.id})" class="text-rose-400 hover:text-rose-300 text-xs font-semibold">Delete</button>
                        </td>
                    </tr>
                `;
            });
        }
